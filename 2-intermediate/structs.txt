package intermediate

import "fmt"

type Person struct {
	firstName string
	lastName  string
	age       int
	address   Address
	PhoneHomeCell
}

type Address struct {
	city    string
	country string
}

type PhoneHomeCell struct {
	home string
	cell string
}

func (p Person) fullName() string {
	return p.firstName + " " + p.lastName
}

func (a *Person) incrementAgeByOne() {
	a.age++
}

func main() {

	p := Person{
		firstName: "John",
		lastName:  "Doe",
		age:       30,
		address: Address{
			city:    "London",
			country: "U.K.",
		},
		PhoneHomeCell: PhoneHomeCell{
			home: "123123123",
			cell: "223232323",
		},
	}

	p1 := Person{
		firstName: "Jane",
		age:       25,
	}
	// p1.address.city = "New York"
	// p1.address.country = "USA"

	p2 := Person{
		firstName: "Jane",
		age:  25,
	}

	fmt.Println(p)
	fmt.Println(p1.lastName)
	fmt.Println(p.fullName())
	fmt.Println(p.address)
	fmt.Println(p1.address.country)
	fmt.Println(p.cell)
	fmt.Println(p.address.city)
	fmt.Println("p and p1 equal? ", p == p1)
	fmt.Println("p1 and p2 equal? ", p1 == p2)

	// Anonymous Structs
	user := struct {
		username string
		email    string
	}{
		username: "user123",
		email:    "pseudoemail@example.com",
	}
	fmt.Println(user.username)

	fmt.Println("Before increment", p.age)
	p.incrementAgeByOne()
	fmt.Println("After increment", p.age)
}
