		}

	fmt.Println(substracter(3))
	fmt.Println(substracter(4))

func adder() func() int {
	i := 0
	fmt.Println("previoues value of i: ", i)
	return func() int {
		i++
		fmt.Println("added 1 to i")
		return i
	}
}
